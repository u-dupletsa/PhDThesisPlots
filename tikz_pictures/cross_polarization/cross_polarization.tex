\documentclass[tufte-handout]{standalone}
\usepackage{mathpazo} % Sets Palatino as the main font
\usepackage{tikz}
\usetikzlibrary{3d, angles, quotes, calc, backgrounds}
\usetikzlibrary{decorations.markings}

% my colors
\definecolor{darkslategray}{rgb}{0.18, 0.31, 0.31}
\definecolor{myorange}{rgb}{0.89804, 0.55686, 0.19608}

\begin{document}

\begin{tikzpicture}

% Define styles
\tikzset{circle grid/.style={myorange, thick, dashed}}

% Parameters
\def\stretch{8} % Amplitude of the stretch
% h× Polarization
\begin{scope}[shift={(5,0)}] % Shift right
  % Draw axes
  \draw[->, thick] (-1.5,0) -- (1.5,0) node[right] {\scriptsize $x$};
  \draw[->, thick] (0,-1.5) -- (0,1.5) node[above] {\scriptsize $y$};
  
  % Draw original circle
  \draw[circle grid] (0,0) circle (1);

  % Draw deformed ellipse (h×)
  \draw[thick, darkslategray] plot[domain=0:360, smooth, samples=100] 
    ({cos(\x) + \stretch * sin(2) * sin(\x)}, {sin(\x) + \stretch * sin(2) * cos(\x)});
  \draw[thick, darkslategray] plot[domain=0:360, smooth, samples=100] 
    ({cos(\x) - \stretch * sin(2) * sin(\x)}, {sin(\x) - \stretch * sin(2) * cos(\x)});
  
  % Title
  \node at (0,-1.8) {\scriptsize $h_\times$};
\end{scope}

\end{tikzpicture}

\end{document}
